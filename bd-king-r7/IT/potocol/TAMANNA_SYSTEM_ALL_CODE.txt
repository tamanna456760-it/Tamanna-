# ==========================================
# TAMANNA ADVANCED SECURITY MONITOR
# Defensive / Monitoring Use Only
# ==========================================

import os
import time
import socket
import psutil
import platform
from datetime import datetime

LOG_DIR = "logs"
os.makedirs(LOG_DIR, exist_ok=True)

LOG_FILE = os.path.join(LOG_DIR, "tamanna_security.log")

# ==========================================
# SYSTEM INFO
# ==========================================

def system_info():

    print("=" * 60)
    print("TAMANNA SECURITY SYSTEM")
    print("=" * 60)

    print("HOSTNAME :", socket.gethostname())
    print("SYSTEM   :", platform.system())
    print("RELEASE  :", platform.release())
    print("PROCESSOR:", platform.processor())

# ==========================================
# CPU / RAM MONITOR
# ==========================================

def monitor_resources():

    cpu = psutil.cpu_percent(interval=1)
    ram = psutil.virtual_memory().percent

    print(f"[CPU] {cpu}%")
    print(f"[RAM] {ram}%")

# ==========================================
# NETWORK MONITOR
# ==========================================

def network_monitor():

    print("\n[ACTIVE CONNECTIONS]")

    connections = psutil.net_connections()

    for conn in connections[:20]:

        try:
            print(conn.laddr)

        except:
            pass

# ==========================================
# PROCESS MONITOR
# ==========================================

def process_monitor():

    print("\n[RUNNING PROCESSES]")

    for proc in psutil.process_iter(['pid', 'name']):

        try:
            print(proc.info)

        except:
            pass

# ==========================================
# LOG SYSTEM
# ==========================================

def write_log(message):

    with open(LOG_FILE, "a") as log:
        log.write(f"{datetime.now()} : {message}\n")

# ==========================================
# SECURITY ALERT
# ==========================================

def security_alert():

    suspicious = ["nc", "netcat", "hydra"]

    for proc in psutil.process_iter(['name']):

        try:

            pname = proc.info['name']

            if pname and pname.lower() in suspicious:

                alert = f"[ALERT] Suspicious Tool Detected: {pname}"

                print(alert)

                write_log(alert)

        except:
            pass

# ==========================================
# MAIN SYSTEM
# ==========================================

def main():

    system_info()

    while True:

        monitor_resources()

        network_monitor()

        process_monitor()

        security_alert()

        write_log("System scan completed")

        print("\nNext scan in 10 seconds...")

        time.sleep(10)

if __name__ == "__main__":
    main()
